# Check Status
git status

# Add Files
git add .

# Commit Changes
git commit -m "Your Message"

# Push Changes
git push

# Pull Latest Changes
git pull

# Check Branches
git branch

# View Commit History
git log --oneline

# Check Remote Repository
git remote -v
